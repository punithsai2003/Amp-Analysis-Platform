    <script>
        const API_URL = 'http://localhost:5000/api';
        let currentJobId = null;
        let currentGenJobId = null;
        let selectedTopN = 10;

        function updateStep(stepNum) {
            for (let i = 1; i <= 5; i++) {
                const circle = document.getElementById(`step${i}`);
                const line = document.getElementById(`line${i}`);
                
                circle.classList.remove('active', 'completed');
                if (line) line.classList.remove('completed');
                
                if (i < stepNum) {
                    circle.classList.add('completed');
                    if (line) line.classList.add('completed');
                } else if (i === stepNum) {
                    circle.classList.add('active');
                }
            }
        }

        // Upload
        document.getElementById('trainingFileInput').addEventListener('change', async (e) => {
            const file = e.target.files[0];
            if (!file) return;

            const formData = new FormData();
            formData.append('file', file);

            document.getElementById('trainingFileName').textContent = '⏳ Uploading...';
            document.getElementById('trainingFileName').classList.remove('hidden');
            document.getElementById('trainingFileName').className = 'text-gray-600 mt-6 font-medium pulse';

            try {
                const response = await fetch(`${API_URL}/upload-training`, {
                    method: 'POST',
                    body: formData
                });

                const data = await response.json();
                
                if (data.success) {
                    currentJobId = data.job_id;
                    document.getElementById('trainingFileName').textContent = 
                        `✅ Loaded ${data.count} training sequences`;
                    document.getElementById('trainingFileName').className = 'text-green-600 mt-6 font-bold text-lg';
                    
                    document.getElementById('generateSection').classList.remove('hidden');
                    updateStep(2);
                    
                    setTimeout(() => {
                        document.getElementById('generateSection').scrollIntoView({ behavior: 'smooth', block: 'center' });
                    }, 300);
                } else {
                    document.getElementById('trainingFileName').textContent = 
                        `❌ Error: ${data.error}`;
                    document.getElementById('trainingFileName').className = 'text-red-600 mt-6 font-medium';
                }
            } catch (error) {
                document.getElementById('trainingFileName').textContent = 
                    `❌ Upload failed: ${error.message}`;
                document.getElementById('trainingFileName').className = 'text-red-600 mt-6 font-medium';
            }
        });

        // Generate
        document.getElementById('generateBtn').addEventListener('click', async () => {
            if (!currentJobId) {
                alert('Please upload training sequences first');
                return;
            }

            const numGenerate = parseInt(document.getElementById('numGenerate').value);
            
            if (numGenerate < 10 || numGenerate > 1000) {
                alert('Please enter between 10 and 1000 sequences');
                return;
            }

            document.getElementById('generateBtn').disabled = true;
            document.getElementById('progressSection').classList.remove('hidden');
            updateStep(3);

            const progressBar = document.getElementById('progressBar');
            const progressPercent = document.getElementById('progressPercent');
            const progressStep = document.getElementById('progressStep');
            const progressLabel = document.getElementById('progressLabel');

            progressLabel.textContent = 'Generating sequences with ProtGPT2...';
            progressStep.textContent = 'Analyzing training patterns and creating new candidates...';

            let progress = 0;
            const interval = setInterval(() => {
                if (progress < 90) {
                    progress += Math.random() * 10;
                    progressBar.style.width = `${Math.min(progress, 90)}%`;
                    progressPercent.textContent = `${Math.floor(Math.min(progress, 90))}%`;
                }
            }, 500);

            try {
                const response = await fetch(`${API_URL}/generate`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        job_id: currentJobId,
                        num_generate: numGenerate
                    })
                });

                const data = await response.json();
                clearInterval(interval);

                if (data.success) {
                    currentGenJobId = data.gen_job_id;
                    
                    progressBar.style.width = '100%';
                    progressPercent.textContent = '100%';
                    progressLabel.textContent = '✅ Generation Complete!';
                    progressStep.textContent = `Successfully generated ${data.total_generated} new sequences`;

                    setTimeout(() => {
                        document.getElementById('progressSection').classList.add('hidden');
                        document.getElementById('screeningSection').classList.remove('hidden');
                        
                        document.getElementById('totalGenerated').textContent = data.total_generated;
                        document.getElementById('highQuality').textContent = 
                            Math.floor(data.total_generated * 0.7);
                        document.getElementById('avgScore').textContent = '0.75';
                        
                        updateStep(3);
                        document.getElementById('screeningSection').scrollIntoView({ behavior: 'smooth', block: 'center' });
                    }, 1000);
                } else {
                    progressLabel.textContent = 'Error during generation';
                    progressStep.textContent = data.error;
                    alert('Generation failed: ' + data.error);
                }
            } catch (error) {
                clearInterval(interval);
                progressLabel.textContent = 'Error';
                progressStep.textContent = error.message;
                alert('Generation failed: ' + error.message);
            } finally {
                document.getElementById('generateBtn').disabled = false;
            }
        });

            btn.addEventListener('click', function() {
                document.querySelectorAll('.top-n-btn').forEach(b => {
                    b.classList.remove('bg-gradient-to-r', 'from-purple-500', 'to-purple-600', 'text-white', 'border-purple-600', 'shadow-lg');
                    b.classList.add('border-purple-400', 'bg-white');

        document.getElementById('analyzeTopNBtn').addEventListener('click', async () => {
            if (!currentGenJobId) {
                alert('Please generate sequences first');
                return;
            }

            document.getElementById('analyzeTopNBtn').disabled = true;
            document.getElementById('progressSection').classList.remove('hidden');
            updateStep(4);

            const progressBar = document.getElementById('progressBar');
            const progressPercent = document.getElementById('progressPercent');
            const progressStep = document.getElementById('progressStep');
            const progressLabel = document.getElementById('progressLabel');

            progressLabel.textContent = `Analyzing TOP ${selectedTopN} candidates...`;
            progressStep.textContent = 'Computing AlphaFold metrics, structure predictions, and functional properties...';

            let progress = 0;
            const interval = setInterval(() => {
                if (progress < 90) {
                    progress += Math.random() * 5;
                    progressBar.style.width = `${Math.min(progress, 90)}%`;
                    progressPercent.textContent = `${Math.floor(Math.min(progress, 90))}%`;
                }
            }, 1000);

            try {
                const response = await fetch(`${API_URL}/analyze-top10`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        gen_job_id: currentGenJobId,
                        top_n: selectedTopN
                    })
                });

                const data = await response.json();
                clearInterval(interval);

                if (data.success) {
                    progressBar.style.width = '100%';
                    progressPercent.textContent = '100%';
                    progressLabel.textContent = '✅ Analysis Complete!';
                    progressStep.textContent = `Successfully analyzed ${data.results.length} sequences with full metrics`;

                    setTimeout(() => {
                        document.getElementById('progressSection').classList.add('hidden');
                        displayResults(data.results);
                        updateStep(5);
                    }, 1000);
                } else {
                    alert('Analysis failed: ' + data.error);
                }
            } catch (error) {
                clearInterval(interval);
                alert('Analysis failed: ' + error.message);
            } finally {
                document.getElementById('analyzeTopNBtn').disabled = false;
            }
        });

        // Display Results
        function displayResults(results) {
            const tbody = document.getElementById('resultsBody');
            tbody.innerHTML = '';

            results.forEach((r, idx) => {
                const row = document.createElement('tr');

                // Score badges with CORRECT thresholds
                // pLDDT: 0-100 scale
                const pLDDT = parseFloat(r.pLDDT) || 0;
                const pLDDTClass = pLDDT >= 70 ? 'metric-high' : pLDDT >= 50 ? 'metric-medium' : 'metric-low';
                
                // pTM and ipTM: 0-1 scale
                const pTM = parseFloat(r.pTM) || 0;
                const pTMClass = pTM >= 0.6 ? 'metric-high' : pTM >= 0.4 ? 'metric-medium' : 'metric-low';
                
                const ipTM = parseFloat(r.ipTM) || 0;
                const ipTMClass = ipTM >= 0.6 ? 'metric-high' : ipTM >= 0.4 ? 'metric-medium' : 'metric-low';
                
                const hasStructure = r.pdb_path && r.pdb_path !== 'None' && r.pdb_path !== null;
                const structureButton = hasStructure 
                    ? `<button onclick="viewStructure('${r.id}')" 
                              class="px-3 py-2 bg-gradient-to-r from-blue-500 to-blue-600 text-white rounded-lg hover:from-blue-600 hover:to-blue-700 text-xs font-semibold shadow-md transition-all">
                          View
                       </button>`
                    : `<span class="text-gray-400 text-xs">N/A</span>`;

                row.innerHTML = `
                    <td class="p-4">
                        <span class="inline-flex items-center justify-center w-8 h-8 rounded-full bg-gradient-to-br from-purple-500 to-purple-600 text-white font-bold text-sm">
                            ${idx + 1}
                        </span>
                    </td>
                    <td class="p-4 font-mono text-sm font-semibold text-gray-800">${r.id}</td>
                    <td class="p-4 font-mono text-xs text-gray-600" title="${r.sequence}">${r.sequence.substring(0, 25)}...</td>
                    <td class="p-4 text-center font-semibold">${r.length}</td>
                    <td class="p-4 text-center">
                        <span class="metric-badge ${pLDDTClass}">${r.pLDDT || 'N/A'}</span>
                    </td>
                    <td class="p-4 text-center">
                        <span class="metric-badge ${pTMClass}">${r.pTM || 'N/A'}</span>
                    </td>
                    <td class="p-4 text-center">
                        <span class="metric-badge ${ipTMClass}">${r.ipTM || 'N/A'}</span>
                    </td>
                    <td class="p-4 text-center text-sm">${r.molecular_weight || 'N/A'}</td>
                    <td class="p-4 text-center text-sm">${r.isoelectric_point || 'N/A'}</td>
                    <td class="p-4 text-center text-sm">${r.helix || 'N/A'}%</td>
                    <td class="p-4 text-center">
                        <span class="metric-badge metric-high">${(r.ampScore || 0).toFixed(2)}</span>
                    </td>
                    <td class="p-4 text-center text-xs">
                        <span class="px-2 py-1 rounded-full ${r.toxic === 'Non-Toxic' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'} font-semibold">
                            ${r.toxic || 'N/A'}
                        </span>
                    </td>
                    <td class="p-4 text-center text-xs">
                        <span class="px-2 py-1 rounded-full ${r.allergen === 'Non-Allergen' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'} font-semibold">
                            ${r.allergen || 'N/A'}
                        </span>
                    </td>
                    <td class="p-4 text-center text-xs">
                        <span class="px-2 py-1 rounded-full ${r.hemolytic === 'Non-Hemolytic' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'} font-semibold">
                            ${r.hemolytic || 'N/A'}
                        </span>
                    </td>
                    <td class="p-4 text-center">${structureButton}</td>
                `;
                
                tbody.appendChild(row);
            });

            document.getElementById('resultsSection').classList.remove('hidden');
            document.getElementById('resultsSection').scrollIntoView({ behavior: 'smooth', block: 'center' });
        }

        async function checkBackendHealth() {

                const response = await fetch(`${API_URL}/health`, {

                    headers: { 'Accept': 'application/json' }
                });
                
                if (!response.ok) throw new Error(`HTTP ${response.status}`);
                
                const data = await response.json();
                console.log('✅ Backend connected:', data);
                
                const header = document.querySelector('.glass-card');
                const banner = document.createElement('div');
                banner.className = 'mt-4 p-3 bg-green-500 bg-opacity-20 border-2 border-green-400 rounded-xl text-green-100 font-semibold text-center fade-in';
                banner.textContent = '✅ Backend server connected successfully!';
                header.appendChild(banner);
                setTimeout(() => banner.remove(), 5000);
                
            } catch (error) {
                console.error('❌ Backend connection failed:', error);
                
                const header = document.querySelector('.glass-card');
                const banner = document.createElement('div');
                banner.className = 'mt-4 p-4 bg-red-500 bg-opacity-20 border-2 border-red-400 rounded-xl text-red-100 fade-in';
